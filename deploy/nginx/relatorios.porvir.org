server {
  listen 80;
  server_name relatorios.porvir.org;

  root /var/www/relatorios_porvir/dist;
  index index.html;

  access_log /var/log/nginx/relatorios_porvir.access.log;
  error_log /var/log/nginx/relatorios_porvir.error.log;

  location /api/ {
    proxy_pass http://127.0.0.1:5102/api/;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
  }

  location / {
    try_files $uri $uri/ /index.html;
  }
}
